# API Configuration
# Backend API base URL, used by the server only (sign-in and the /api/admin forwarder).
# The browser calls this app's own /api/admin/... routes and never holds the API token.
API_URL=http://localhost:8000/api/v1
# Still read as a fallback when API_URL is not set.
NEXT_PUBLIC_API_BASE_URL=http://localhost:8000/api/v1

# GraphQL (optional - for FastAPI, Strapi, etc.), forwarded through /api/graphql
GRAPHQL_ENDPOINT=http://localhost:8000/graphql

# Authentication
# gpp writes a real NEXTAUTH_SECRET into .env.local. The dev script sets
# NEXTAUTH_URL to the port Next binds (PORT, or 3000). NextAuth otherwise
# assumes http://localhost:3000, and the session cookie does not match.
NEXTAUTH_URL=http://localhost:3000
NEXTAUTH_SECRET=your-secret-key-change-in-production

# Prisma Schema (for CRUD generation)
# If using shared schema in monorepo:
PRISMA_SCHEMA_PATH=../packages/prisma-schema/schema.prisma

# Site Configuration
NEXT_PUBLIC_SITE_NAME=Admin Dashboard
