# Server Configuration
HOST=0.0.0.0
PORT=8000
DEBUG=false
# development lets the app start with the placeholder JWT_SECRET below.
# Set ENVIRONMENT=production (the default when unset) and a real JWT_SECRET
# when deploying: production refuses to start with a "change-me" secret.
ENVIRONMENT=development

# Database. gpp also writes DATABASE_URL into this folder's .env, which is what
# the app and the Alembic CLI read.
# Local: localhost. Inside the project's docker compose the host is the DB
# service name: postgresql+psycopg://postgres:postgres@{{PROJECT_SLUG}}-db:5432/postgres
DATABASE_URL=postgresql+psycopg://postgres:postgres@localhost:5432/postgres

# Authentication. Generate a real secret with:
#   python -c "import secrets; print(secrets.token_urlsafe(48))"
JWT_SECRET=change-me-in-production-use-a-strong-secret-key
JWT_ALGORITHM=HS256
ACCESS_TOKEN_EXPIRE_MINUTES=60
REFRESH_TOKEN_EXPIRE_DAYS=30

# CORS (comma-separated exact origins). A gpp Electron app adds
# http://localhost:5180 (dev) and app://bundle (built app). Don't add "null";
# see README "CORS".
CORS_ORIGINS=http://localhost:3000,http://localhost:3001

# Redis (leave empty, i.e. "REDIS_URL=", to disable; /health then reports "redis": "disabled")
# Local: localhost. Inside the project's docker compose: redis://{{PROJECT_SLUG}}-redis:6379/0
REDIS_URL=redis://localhost:6379/0
