# Server Configuration
NODE_ENV=development
API_PORT=3001

# SQLite file next to prisma/schema.prisma. A packaged Electron app overrides
# DATABASE_URL with userData/app.db.
DATABASE_URL="file:./dev.db"

# JWT Authentication. With NODE_ENV=production the server refuses to start unless
# JWT_SECRET is set, is not a placeholder (nothing containing "change-me") and is at
# least 32 characters, e.g. `openssl rand -hex 32`.
JWT_SECRET=gpp:generate-secret
# Access tokens are stateless (valid until they expire). Refresh tokens are stored as
# hashes in the refresh_tokens table, rotated on every /auth/refresh and revoked by
# /auth/logout.
JWT_EXPIRES_IN=1h
REFRESH_TOKEN_EXPIRES_IN=30d
# Seconds a just-replaced refresh token still works (0 = never).
REFRESH_TOKEN_REUSE_GRACE_SECONDS=30

# CORS (comma-separated origins for production)
CORS_ORIGINS=http://localhost:3000,http://localhost:3001
